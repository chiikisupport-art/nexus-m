    function parseDuration(t) {
      let ms = 0, hit = false;
      const h = t.match(/(\d+(?:\.\d+)?)\s*時間/); if (h) { ms += h[1] * 3600000; hit = true; }
      const m = t.match(/(\d+(?:\.\d+)?)\s*分(?!の)/); if (m) { ms += m[1] * 60000; hit = true; }
      const s = t.match(/(\d+)\s*秒/); if (s) { ms += s[1] * 1000; hit = true; }
      if (/半/.test(t) && h) ms += 1800000;
      return hit ? ms : null;
    }

    function calc(expr) {
      const tokens = expr.match(/\d+(?:\.\d+)?|[-+*/^%()]|√|sqrt/g);
      if (!tokens || tokens.join("") !== expr.replace(/\s+/g, "")) throw new Error("bad");
      let i = 0;
      const peek = () => tokens[i], next = () => tokens[i++];
      function primary() {
        const t = next();
        if (t === "(") { const v = add(); if (next() !== ")") throw new Error("paren"); return v; }
        if (t === "-") return -primary();
        if (t === "+") return primary();
        if (t === "√" || t === "sqrt") return Math.sqrt(primary());
        if (t !== undefined && /^\d/.test(t)) return parseFloat(t);
        throw new Error("token");
      }
      function power() { let v = primary(); if (peek() === "^") { next(); v = Math.pow(v, power()); } return v; }
      function mul() {
        let v = power();
        while (["*", "/", "%"].includes(peek())) { const op = next(), r = power(); v = op === "*" ? v * r : op === "/" ? v / r : v % r; }
        return v;
      }
      function add() {
        let v = mul();
        while (["+", "-"].includes(peek())) { const op = next(), r = mul(); v = op === "+" ? v + r : v - r; }
        return v;
      }
      const v = add();
      if (i !== tokens.length) throw new Error("trail");
      return v;
    }

        .replace(/かける|掛ける|×|x(?=\d)|X(?=\d)/g, "*").replace(/わる|割る|÷/g, "/")
        .replace(/の(\d+)乗/g, "^$1").replace(/の二乗|の2乗/g, "^2")
        .replace(/\s+/g, "").trim();
    }
